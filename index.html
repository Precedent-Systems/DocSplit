DocSplit CLI Integration

# 1. Install DocSplit locally
git clone https://github.com/Precedent-Systems/DocSplit.git
cd DocSplit
pip install -e .

# 2. Slice a court record PDF (with automatic OCR health warnings)
docsplit split "/path/to/24CV21417_record.pdf"

# 3. Slice and auto-repair missing/broken OCR pages in one pass
docsplit split "/path/to/24CV21417_record.pdf" --repair-ocr

# 4. Audit an existing folder of sliced PDFs for OCR or handwriting gaps
docsplit audit "/path/to/02_Case_24CV21417_Clackamas_Quiet_Title"

# 5. Repair flagged pages (with optional curated handwriting overlay manifest)
docsplit repair "/path/to/02_Case_24CV21417_Clackamas_Quiet_Title" --overlay-json overlays.json
